"""
Parse job notifications from @OffersHunterBot messages.

The bot sends Markdown-formatted messages. Common fields:

  📌 *Title*
  💰 Budget: $15–25/hr  or  Fixed: $500
  🌍 Country: United States
  💵 Client spent: $10k+
  📅 Posted: 2 hours ago
  📝 Description text...
  🔗 https://www.upwork.com/jobs/~01abc...

Field labels vary slightly between message versions, so we use
multiple regex patterns per field.
"""

import re
from urllib.parse import urlparse


# --- helpers -----------------------------------------------------------------

def _first(patterns: list[str], text: str, flags=re.IGNORECASE) -> str | None:
    for pat in patterns:
        m = re.search(pat, text, flags)
        if m:
            return m.group(1).strip()
    return None


def _extract_upwork_id(url: str) -> str | None:
    if not url:
        return None
    m = re.search(r"~([0-9a-f]{16,})", url)
    return m.group(1) if m else None
